cmake_minimum_required(VERSION 3.25)
project(AstralLocalizedDamageTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(ASTRAL_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
include(CTest)
if(BUILD_TESTING)
    include("${ASTRAL_SOURCE_ROOT}/cmake/AstralTestSafety.cmake")
    add_executable(LocalizedDamageTests LocalizedDamageTests.cpp)
    target_include_directories(LocalizedDamageTests PRIVATE "${ASTRAL_SOURCE_ROOT}")
    if(MSVC)
        target_compile_options(LocalizedDamageTests PRIVATE /W4 /WX)
    else()
        target_compile_options(LocalizedDamageTests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    astral_add_test(LocalizedDamageTests)
endif()
