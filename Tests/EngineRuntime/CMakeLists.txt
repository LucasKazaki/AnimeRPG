cmake_minimum_required(VERSION 3.25)
project(AstralEngineRuntimePortable LANGUAGES CXX)

# Headless portable build of the REAL engine runtime sources and their suites.
# It is not a build of AstralGame (Win32) and does not replace Windows CTest,
# native interactive smokes, GPU/performance evidence or the required soak.
get_filename_component(ASTRAL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ASTRAL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(ASTRAL_TSAN "Enable the thread sanitizer" OFF)
if(ASTRAL_SANITIZE AND ASTRAL_TSAN)
    message(FATAL_ERROR "ASTRAL_SANITIZE and ASTRAL_TSAN are mutually exclusive")
endif()

include("${ASTRAL_ROOT}/cmake/AstralEngineSources.cmake")
include("${ASTRAL_ROOT}/cmake/AstralRuntime.cmake")

function(astral_portable_options target)
    target_include_directories(${target} PRIVATE "${ASTRAL_ROOT}")
    if(MSVC)
        # Warnings-as-errors is enforced on the GCC/Clang legs; MSVC /W4 adds
        # conversion warnings those compilers do not report.
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    if(ASTRAL_SANITIZE)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer
            -fno-sanitize-recover=all)
        target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
    if(ASTRAL_TSAN)
        target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE -fsanitize=thread)
    endif()
endfunction()

astral_add_runtime()
astral_portable_options(AstralEngineRuntime)

add_library(AstralShowcase STATIC ${ASTRAL_SHOWCASE_SOURCES} ${ASTRAL_GAMEPLAY_SOURCES} ${ASTRAL_SAMPLE_SOURCES})
astral_portable_options(AstralShowcase)
target_link_libraries(AstralShowcase PUBLIC AstralEngineRuntime)

add_library(AstralEditorCore STATIC ${ASTRAL_EDITOR_SOURCES})
astral_portable_options(AstralEditorCore)
target_link_libraries(AstralEditorCore PUBLIC AstralEngineRuntime)

if(EXISTS "${ASTRAL_ROOT}/Tools/AstralCapture.cpp")
    add_executable(AstralCapture "${ASTRAL_ROOT}/Tools/AstralCapture.cpp")
    astral_portable_options(AstralCapture)
    target_link_libraries(AstralCapture PRIVATE AstralShowcase)
endif()
add_executable(AstralPlayer "${ASTRAL_ROOT}/Tools/AstralPlayer.cpp")
astral_portable_options(AstralPlayer)
target_compile_definitions(AstralPlayer PRIVATE ASTRAL_PLAYER_PLAYGROUND)
target_link_libraries(AstralPlayer PRIVATE AstralShowcase)
add_executable(AstralEnginePlayer "${ASTRAL_ROOT}/Tools/AstralPlayer.cpp")
astral_portable_options(AstralEnginePlayer)
target_link_libraries(AstralEnginePlayer PRIVATE Astral::Runtime)
add_executable(AstralProject "${ASTRAL_ROOT}/Tools/AstralProject.cpp")
astral_portable_options(AstralProject)
target_link_libraries(AstralProject PRIVATE Astral::Runtime)

include(CTest)
if(BUILD_TESTING)
    include("${ASTRAL_ROOT}/cmake/AstralTestSafety.cmake")
    foreach(suite IN LISTS ASTRAL_ENGINE_TEST_SUITES)
        add_executable(${suite} "${ASTRAL_ROOT}/Tests/${suite}.cpp")
        astral_portable_options(${suite})
        if(suite STREQUAL "EngineProjectTests")
            target_link_libraries(${suite} PRIVATE Astral::Runtime)
        else()
            target_link_libraries(${suite} PRIVATE AstralShowcase)
        endif()
        if(suite STREQUAL "EngineEditorTests")
            target_link_libraries(${suite} PRIVATE AstralEditorCore)
        endif()
        # Suites that read sample content find it here.
        target_compile_definitions(${suite} PRIVATE ASTRAL_SOURCE_ROOT="${ASTRAL_ROOT}")
        if(suite STREQUAL "EngineProjectTests")
            astral_add_test(${suite})
        else()
            add_test(NAME ${suite} COMMAND ${suite})
        endif()
        set_tests_properties(${suite} PROPERTIES TIMEOUT 300)
    endforeach()
endif()
